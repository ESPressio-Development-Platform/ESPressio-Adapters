cmake_minimum_required(VERSION 3.16)
project(ESPressioAdaptersContracts LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(ESPRESSIO_DEPENDENCIES "${CMAKE_CURRENT_LIST_DIR}/.." CACHE PATH "Sibling ESPressio dependency checkout root")
set(ADAPTER_INCLUDES "${CMAKE_CURRENT_LIST_DIR}/../src")
foreach(dependency System Primitive Task)
    list(APPEND ADAPTER_INCLUDES "${ESPRESSIO_DEPENDENCIES}/ESPressio-${dependency}/src")
endforeach()

file(GLOB ADAPTER_PUBLIC_HEADERS "${CMAKE_CURRENT_LIST_DIR}/../src/*.hpp")
set(FORBIDDEN_TOKENS ESPressio_Event ESPressio_Command ESPressio_State ESPressio_Mesh ESPressio_Radio ESPressio_Threads ESPressio_Timing ESPressio_Serializable ESPressio_Persistence ESPressio_Observable std::function)
foreach(header IN LISTS ADAPTER_PUBLIC_HEADERS)
    file(READ "${header}" text)
    foreach(token IN LISTS FORBIDDEN_TOKENS)
        string(FIND "${text}" "${token}" index)
        if(NOT index EQUAL -1)
            message(FATAL_ERROR "Forbidden generic-Adapters dependency/token ${token} found in ${header}")
        endif()
    endforeach()
endforeach()

enable_testing()
foreach(name byte_arena capacity capacity_rollback capacity_fit no_heap)
    add_executable(adapter_${name} test_adapter_${name}.cpp)
    target_include_directories(adapter_${name} PRIVATE ${ADAPTER_INCLUDES})
    target_compile_definitions(adapter_${name} PRIVATE ESPRESSIO_ADAPTERS_TESTING=1)
    target_compile_options(adapter_${name} PRIVATE -Wall -Wextra -Werror -fno-rtti -UNDEBUG)
    add_test(NAME adapter_${name} COMMAND adapter_${name})
endforeach()

set(COMPILE_INCLUDES "-I${CMAKE_CURRENT_LIST_DIR}/../src")
foreach(dependency System Primitive Task)
    list(APPEND COMPILE_INCLUDES "-I${ESPRESSIO_DEPENDENCIES}/ESPressio-${dependency}/src")
endforeach()
add_test(NAME adapter_compile_fail_byte_lease_copy
    COMMAND ${CMAKE_CXX_COMPILER} -std=c++17 -fno-rtti -Wall -Wextra -Werror -fsyntax-only ${COMPILE_INCLUDES}
            "${CMAKE_CURRENT_LIST_DIR}/negative/byte_lease_copy.cpp")
set_tests_properties(adapter_compile_fail_byte_lease_copy PROPERTIES WILL_FAIL TRUE)
